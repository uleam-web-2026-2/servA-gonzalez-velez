package main

import (
	"log"
	"net/http"

	"github.com/go-chi/chi/v5"
	"github.com/go-chi/chi/v5/middleware"
	"gorm.io/driver/postgres"
	"gorm.io/gorm"

	"canchas/internal/almacen"
	"canchas/internal/config"
	"canchas/internal/reservas"
	"canchas/internal/sesion"
	"canchas/internal/transporte"
)

func main() {
	// 1. Leer el entorno. Si falta algo, se detiene aquí, con un mensaje claro.
	cfg, err := config.Cargar()
	if err != nil {
		log.Fatal("configuración: ", err)
	}

	// 2. Conectar y migrar.
	db, err := gorm.Open(postgres.Open(cfg.DatabaseURL), &gorm.Config{})
	if err != nil {
		log.Fatal("no se pudo conectar: ", err)
	}
	err = db.AutoMigrate(&reservas.Cancha{}, &reservas.Reserva{}, &sesion.Usuario{})
	if err != nil {
		log.Fatal("no se pudo migrar: ", err)
	}
	if err := almacen.Sembrar(db); err != nil {
		log.Fatal("no se pudo sembrar: ", err)
	}
	if err := sesion.Sembrar(db); err != nil {
		log.Fatal("no se pudieron sembrar los usuarios: ", err)
	}

	// 3. Rutas. El secreto de la firma llega de la configuración.
	r := chi.NewRouter()
	r.Use(middleware.Logger)
	r.Use(middleware.Recoverer)
	r.Use(middleware.Timeout(cfg.TiempoEspera))
	tokens := sesion.NuevosTokens(cfg.ClaveFirma)
	login := &sesion.Manejador{DB: db, Tokens: tokens}
	r.Post("/login", login.Login) // pública: es la puerta para conseguir el token
	repo := &almacen.Reservas{DB: db}
	servicio := reservas.NuevoServicio(repo)
	(&transporte.Reservas{Servicio: servicio}).Rutas(r, tokens)

	// 4. Servidor: el tiempo de espera se escribe UNA vez, en la configuración.
	servidor := &http.Server{
		Addr:         ":" + cfg.Puerto,
		Handler:      r,
		ReadTimeout:  cfg.TiempoEspera,
		WriteTimeout: cfg.TiempoEspera,
	}
	log.Println("escuchando en el puerto", cfg.Puerto)
	log.Fatal(servidor.ListenAndServe())
}
